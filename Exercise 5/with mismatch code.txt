#include <cstdio>
#include <mpi.h>

int main(int argc, char **argv)
{
    MPI_Init(&argc, &argv);

    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size < 4) {
        if (rank == 0)
            printf("Run with at least 4 processes.\n");
        MPI_Finalize();
        return 1;
    }

    int x = 30;
    int y;
    MPI_Status status;

    if (rank == 1) {
        printf("Rank 1: sending to rank 3\n");
        fflush(stdout);

        MPI_Ssend(&x, 1, MPI_INT, 3, 0, MPI_COMM_WORLD);
    }
    else if (rank == 3) {
        printf("Rank 3: waiting for rank 2 (intentional mismatch)\n");
        fflush(stdout);

        // Wrong source: rank 2 instead of rank 1.
        MPI_Recv(&y, 1, MPI_INT, 2, 0,
                 MPI_COMM_WORLD, &status);

        printf("Received y = %d\n", y);
    }

    MPI_Finalize();
    return 0;
}
