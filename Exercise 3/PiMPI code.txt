#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <mpi.h>

int main(int argc, char **argv)
{
    MPI_Init(&argc, &argv);

    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    const long long totalTrials = 10000000;

    /* Distribute all trials, including any remainder. */
    long long localTrials = totalTrials / size;
    if (rank < totalTrials % size) {
        localTrials++;
    }

    /* Give each process a different random seed. */
    unsigned int seed = (unsigned int)time(NULL)
                      + 12345u * (unsigned int)rank;
    srand(seed);

    MPI_Barrier(MPI_COMM_WORLD);
    double startTime = MPI_Wtime();

    long long localInside = 0;

    for (long long i = 0; i < localTrials; i++) {
        double x = (double)rand() / ((double)RAND_MAX + 1.0);
        double y = (double)rand() / ((double)RAND_MAX + 1.0);

        if (x * x + y * y <= 1.0) {
            localInside++;
        }
    }

    long long totalInside = 0;

    MPI_Reduce(
        &localInside, &totalInside, 1,
        MPI_LONG_LONG_INT, MPI_SUM,
        0, MPI_COMM_WORLD
    );

    double elapsed = MPI_Wtime() - startTime;
    double maxElapsed = 0;

    MPI_Reduce(
        &elapsed, &maxElapsed, 1,
        MPI_DOUBLE, MPI_MAX,
        0, MPI_COMM_WORLD
    );

    char name[MPI_MAX_PROCESSOR_NAME];
    int length;
    MPI_Get_processor_name(name, &length);

    printf("Rank %d on %s: trials = %lld, inside = %lld\n",
           rank, name, localTrials, localInside);

    if (rank == 0) {
        double pi = 4.0 * (double)totalInside / totalTrials;

        printf("Total trials = %lld\n", totalTrials);
        printf("Estimated Pi = %.10f\n", pi);
        printf("Elapsed time = %.6f seconds\n", maxElapsed);
    }

    MPI_Finalize();
    return 0;
}
